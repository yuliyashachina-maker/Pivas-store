```ts
types.setTypeParser(types.builtins.NUMERIC, (v) => Number(v));
types.setTypeParser(types.builtins.INT8, (v) => Number(v));
types.setTypeParser(types.builtins.DATE, (v) => v);

type Env = { DATABASE_URL: string };
const app = new Hono<{ Bindings: Env }>();

app.get("/app-api/products", async (c) => {
	const sql = neon(c.env.DATABASE_URL);
	return c.json(await sql`SELECT "id"::text AS "id", "name", "category", "price", "stock", "photo", "description" FROM "products" ORDER BY "id" DESC`);

app.post("/app-api/products", async (c) => {
	const b = await c.req.json();
	const sql = neon(c.env.DATABASE_URL);
	const [row] = await sql`INSERT INTO "products" ("name", "category", "price", "stock", "photo", "description") VALUES (${b.name ?? null}, ${b.category === "" ? null : b.category ?? null}, ${b.price === "" ? null : b.price ?? null}, ${b.stock === "" ? null : b.stock ?? null}, ${b.photo ?? null}, ${b.description ?? null}) RETURNING "id"::text AS "id", "name", "category", "price", "stock", "photo", "description"`;
```

```ts
// PIVAS Beer & Snack data: each hook reads or changes one table of the project database.
import { useQuery, useQueryClient } from "@tanstack/react-query";

export const TODAY = new Date().toISOString().slice(0, 10);

export interface Product {
  id: string;
  name: string;
  category: "Beer" | "Snacks";
  price: number;
  stock: number;
  photo: string;
  description: string;
}
export interface Order {
  id: string;
  customerName: string;
  phone: string;
  address: string;
  deliveryWindow: string;
  notes: string;
  total: number;
  status: "Pending" | "Preparing" | "Out for delivery" | "Delivered";
  createdAt: string;
}

function useRecords<T>(table: string): T[] {
	const { data } = useQuery({ queryKey: [table], queryFn: () => fetch(`/app-api/${table}`).then((r) => r.json()) });
	return (data as T[] | undefined) ?? [];
}

/** A write to one table's routes, then a refresh of that table's reads. */
function useWrite(table: string): (method: "POST" | "PATCH" | "DELETE", id: string | null, body?: unknown) => Promise<unknown> {
	const client = useQueryClient();
	return async (method, id, body) => {
		const r = await fetch(`/app-api/${table}${id === null ? "" : `/${encodeURIComponent(id)}`}`, {
			method,
			...(body === undefined ? {} : { headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) }),
		});
		if (!r.ok) throw new Error(`${method} /app-api/${table} failed: ${r.status}`);
		await client.invalidateQueries({ queryKey: [table] });
		return r.status === 204 ? null : r.json();
	};
}

export function useProducts(): Product[] {
	return useRecords<Product>("products");
}
export function useCreateProduct(): (input: Omit<Product, "id">) => Promise<Product> {
	const write = useWrite("products");
	return (input) => write("POST", null, input) as Promise<Product>;
}
export function useUpdateProduct(): (id: string, changes: Partial<Product>) => Promise<void> {
	const write = useWrite("products");
	return async (id, changes) => {
		await write("PATCH", id, changes);
	};
}
export function useDeleteProduct(): (id: string) => Promise<void> {
	const write = useWrite("products");
	return async (id) => {
		await write("DELETE", id);
	};
}

export function useOrders(): Order[] {
	return useRecords<Order>("orders");
}
export function useCreateOrder(): (input: Omit<Order, "id">) => Promise<Order> {
	const write = useWrite("orders");
	return (input) => write("POST", null, input) as Promise<Order>;
}
export function useUpdateOrder(): (id: string, changes: Partial<Order>) => Promise<void> {
	const write = useWrite("orders");
	return async (id, changes) => {
		await write("PATCH", id, changes);
	};
}
export function useDeleteOrder(): (id: string) => Promise<void> {
	const write = useWrite("orders");
	return async (id) => {
		await write("DELETE", id);
	};
}
```

## 5. Customer store: client/pages/Shop.tsx

```tsx
import { useState } from "react";
import { Plus, Minus, ShoppingCart, PackageX } from "lucide-react";
import { PageHeader } from "../components/PageHeader";
import { Input } from "../components/Input";
import { Button } from "../components/Button";
import { Badge } from "../components/Badge";
import { Grid, GridItem } from "../components/Grid";
import { EmptyState } from "../components/EmptyState";
import { toast } from "../components/Toast";
import { useProducts } from "../data";
import type { Product } from "../data";

type Category = "All" | "Beer" | "Snacks";
const CATEGORIES: Category[] = ["All", "Beer", "Snacks"];

export default function ShopPage() {
  const products = useProducts();
  const [query, setQuery] = useState("");
  const [category, setCategory] = useState<Category>("All");
  const [cart, setCart] = useState<Record<string, number>>({});

  const q = query.trim().toLowerCase();
  const visible = products.filter(
    (item) =>
      (category === "All" || item.category === category) &&
      (q === "" || item.name.toLowerCase().includes(q) || item.description.toLowerCase().includes(q)),
  );

  const cartCount = Object.values(cart).reduce((sum, n) => sum + n, 0);

  function addToCart(product: Product) {
    setCart((prev) => {
      const current = prev[product.id] ?? 0;
      if (current >= product.stock) return prev;
      return { ...prev, [product.id]: current + 1 };
    });
    toast("Added to cart", { description: product.name });
  }

  function removeFromCart(product: Product) {
    setCart((prev) => {
      const current = prev[product.id] ?? 0;
      if (current <= 1) {
        const { [product.id]: _removed, ...rest } = prev;
        return rest;
      }
      return { ...prev, [product.id]: current - 1 };
    });
  }

  return (
    <div className="space-y-4 md:space-y-6">
      <PageHeader
        title="Shop"
        actions={
          <Button variant="primary" size="md" asChild>
            <a href="/cart">
              <ShoppingCart /> Cart ({cartCount})
            </a>
          </Button>
        }
      />

      <div className="space-y-3">
        <Input
          type="search"
          aria-label="Search products"
          placeholder="Search beer or snacks…"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
        />
        <div className="flex gap-2 overflow-x-auto pb-1">
          {CATEGORIES.map((c) => (
            <Button
              key={c}
              size="sm"
              variant={category === c ? "primary" : "secondary"}
              aria-pressed={category === c}
              onClick={() => setCategory(c)}
              className="shrink-0"
            >
              {c}
            </Button>
          ))}
        </div>
      </div>

      <Grid
        columns={2}
        ratio="4:3"
        frame="bare"
        emptyState={
          <EmptyState
            icon={<PackageX />}
            title="No products match"
            description="Try another search or category."
            action={
              <Button
                variant="secondary"
                onClick={() => {
                  setQuery("");
                  setCategory("All");
                }}
              >
                Clear filters
              </Button>
            }
          />
        }
      >
        {visible.map((product) => {
          const inCart = cart[product.id] ?? 0;
          const outOfStock = product.stock === 0;
          const atLimit = inCart >= product.stock;
          return (
            <GridItem
              key={product.id}
              prompt={product.photo}
              style="photo"
              ratio="4:3"
              frame="bare"
              title={product.name}
              description={
                <span className="flex flex-wrap items-center gap-1.5">
                  <span className="font-semibold text-primary">${product.price.toFixed(2)}</span>
                  {outOfStock ? (
                    <Badge tone="error">Sold out</Badge>
                  ) : product.stock <= 5 ? (
                    <Badge tone="warning">{product.stock} left</Badge>
                  ) : null}
                </span>
              }
              trailing={
                outOfStock ? (
                  <Button size="sm" variant="secondary" disabled>
                    Unavailable
                  </Button>
                ) : inCart > 0 ? (
                  <div className="flex items-center gap-1">
                    <Button size="icon" variant="secondary" aria-label={`Remove one ${product.name}`} onClick={() => removeFromCart(product)}>
                      <Minus />
                    </Button>
                    <span className="min-w-6 text-center text-sm font-medium">{inCart}</span>
                    <Button size="icon" variant="secondary" aria-label={`Add one ${product.name}`} disabled={atLimit} onClick={() => addToCart(product)}>
                      <Plus />
                    </Button>
                  </div>
                ) : (
                  <Button size="sm" variant="primary" onClick={() => addToCart(product)}>
                    <Plus /> Add
                  </Button>
                )
              }
            />
          );
        })}
      </Grid>
    </div>
  );
}
```

## 6. Customer store: client/pages/Cart.tsx

```tsx
import { useState } from "react";
import { Minus, Plus, Trash2, ShoppingBag } from "lucide-react";
import { Button } from "../components/Button";
import { Badge } from "../components/Badge";
import { List, ListItem } from "../components/List";
import { Media } from "../components/Media";
import { EmptyState } from "../components/EmptyState";
import { PageHeader } from "../components/PageHeader";
import { Card } from "../components/Card";
import { useProducts } from "../data";

interface CartLine {
  productId: string;
  qty: number;
}

export default function CartPage() {
  const products = useProducts();
  const [lines, setLines] = useState<CartLine[]>(() =>
    products.slice(0, 2).map((p) => ({ productId: p.id, qty: 1 }))
  );

  const items = lines
    .map((line) => {
      const product = products.find((p) => p.id === line.productId);
      return product ? { product, qty: line.qty } : null;
    })
    .filter((item) => item !== null);

  const total = items.reduce((sum, item) => sum + item.product.price * item.qty, 0);
  const count = items.reduce((sum, item) => sum + item.qty, 0);

  function changeQty(productId: string, delta: number) {
    setLines((prev) =>
      prev
        .map((line) =>
          line.productId === productId ? { ...line, qty: line.qty + delta } : line
        )
        .filter((line) => line.qty > 0)
    );
  }

  function removeLine(productId: string) {
    setLines((prev) => prev.filter((line) => line.productId !== productId));
  }

  return (
    <div className="space-y-4">
      <PageHeader title="Cart" />

      {items.length === 0 ? (
        <EmptyState
          icon={<ShoppingBag />}
          title="Your cart is empty"
          description="Add beers and snacks from the shop to get started."
          action={
            <Button asChild variant="primary">
              <a href="/shop">Browse shop</a>
            </Button>
          }
        />
      ) : (
        <>
          <List>
            {items.map(({ product, qty }) => (
              <div key={product.id} className="flex items-center gap-3 py-3">
                <Media prompt={product.photo} style="photo" alt={product.name} size="sm" />
                <div className="min-w-0 flex-1">
                  <p className="truncate text-sm font-medium text-primary">{product.name}</p>
                  <p className="mt-0.5 text-xs text-secondary">
                    ${product.price.toFixed(2)} each
                  </p>
                  <div className="mt-2 flex items-center gap-2">
                    <Button
                      size="icon"
                      variant="secondary"
                      aria-label={`Decrease ${product.name}`}
                      onClick={() => changeQty(product.id, -1)}
                    >
                      <Minus />
                    </Button>
                    <span className="min-w-6 text-center text-sm font-medium text-primary">
                      {qty}
                    </span>
                    <Button
                      size="icon"
                      variant="secondary"
                      aria-label={`Increase ${product.name}`}
                      disabled={qty >= product.stock}
                      onClick={() => changeQty(product.id, 1)}
                    >
                      <Plus />
                    </Button>
                    <Button
                      size="icon"
                      variant="ghost-destructive"
                      aria-label={`Remove ${product.name}`}
                      onClick={() => removeLine(product.id)}
                    >
                      <Trash2 />
                    </Button>
                  </div>
                </div>
                <div className="shrink-0 text-right">
                  <p className="text-sm font-semibold text-primary">
                    ${(product.price * qty).toFixed(2)}
                  </p>
                  {qty >= product.stock && (
                    <Badge tone="warning" className="mt-1">
                      Max stock
                    </Badge>
                  )}
                </div>
              </div>
            ))}
          </List>

          <Card>
            <div className="flex items-center justify-between text-sm">
              <span className="text-secondary">
                {count} {count === 1 ? "item" : "items"}
              </span>
              <span className="text-secondary">Subtotal</span>
            </div>
            <div className="mt-2 flex items-center justify-between">
              <span className="text-base font-semibold text-primary">Total</span>
              <span className="text-lg font-semibold text-accent-ink">
                ${total.toFixed(2)}
              </span>
            </div>
            <Button asChild variant="primary" size="lg" className="mt-4 w-full">
              <a href="/checkout">Proceed to checkout</a>
            </Button>
          </Card>
        </>
      )}
    </div>
  );
}
```

```tsx
];

  const createOrder = useCreateOrder();
  const [name, setName] = useState("");
  const [phone, setPhone] = useState("");
  const [address, setAddress] = useState("");
  const [deliveryWindow, setDeliveryWindow] = useState("");
  const [notes, setNotes] = useState("");
  const [errors, setErrors] = useState<Record<string, string>>({});
  const [submitting, setSubmitting] = useState(false);

    const next: Record<string, string> = {};
    if (!name.trim()) next.name = "Enter your name";
```

```ts
import { useQuery, useQueryClient } from "@tanstack/react-query";

export const TODAY = new Date().toISOString().slice(0, 10);

export interface Order {
	id: string;
	customerName: string | null;
	phone: string | null;
	address: string | null;
	deliveryWindow: string | null;
	notes: string | null;
	total: number | null;
	status: string | null;
}
export interface Product {
	id: string;
	name: string | null;
	category: string | null;
	price: number | null;
	stock: number | null;
	photo: string | null;
	description: string | null;
}

function useRecords<T>(table: string): T[] {
	const { data } = useQuery({ queryKey: [table], queryFn: () => fetch(`/app-api/${table}`).then((r) => r.json()) });
	return (data as T[] | undefined) ?? [];
}

/** A write to one table's routes, then a refresh of that table's reads. */
function useWrite(table: string): (method: "POST" | "PATCH" | "DELETE", id: string | null, body?: unknown) => Promise<unknown> {
	const client = useQueryClient();
	return async (method, id, body) => {
		const r = await fetch(`/app-api/${table}${id === null ? "" : `/${encodeURIComponent(id)}`}`, {
			method,
			...(body === undefined ? {} : { headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) }),
		});
		if (!r.ok) throw new Error(`${method} /app-api/${table} failed: ${r.status}`);
		await client.invalidateQueries({ queryKey: [table] });
		return r.status === 204 ? null : r.json();
	};
}

export function useOrders(): Order[] {
	return useRecords<Order>("orders");
}
export function useCreateOrder(): (input: Omit<Order, "id">) => Promise<Order> {
	const write = useWrite("orders");
	return (input) => write("POST", null, input) as Promise<Order>;
}
export function useUpdateOrder(): (id: string, changes: Partial<Order>) => Promise<void> {
	const write = useWrite("orders");
	return async (id, changes) => {
		await write("PATCH", id, changes);
	};
}
export function useDeleteOrder(): (id: string) => Promise<void> {
	const write = useWrite("orders");
	return async (id) => {
		await write("DELETE", id);
	};
}

export function useProducts(): Product[] {
	return useRecords<Product>("products");
}
export function useCreateProduct(): (input: Omit<Product, "id">) => Promise<Product> {
	const write = useWrite("products");
	return (input) => write("POST", null, input) as Promise<Product>;
}
export function useUpdateProduct(): (id: string, changes: Partial<Product>) => Promise<void> {
```

```tsx
function formatWindow(w: string | null): string {
  if (!w) return "No window set";
  const [day, time] = w.split(" ");
  const d = new Date(`${day}T00:00:00`);
  const label = d.toLocaleDateString("en-US", { weekday: "short", month: "short", day: "numeric" });
  return `${label} · ${time}`;
}

function formatMoney(n: number | null): string {
  return n == null ? "—" : `$${n.toFixed(2)}`;
}

export default function OrdersPage() {
  const navigate = useNavigate();
  const orders = useOrders();
  const queryClient = useQueryClient();
  const [filter, setFilter] = useState<Filter>("all");

  const newOrder = () => navigate("/orders/new");

  const setStatus = async (id: string, status: string) => {
    await fetch(`/app-api/orders/${id}`, {
      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ status }),
    });
    queryClient.invalidateQueries({ queryKey: ["orders"] });
  };

  return (
    <div className="space-y-4 md:space-y-6">
      <PageHeader
        title="Orders"
        actions={

          </Button>
        }
      />

      <div className="flex gap-2 overflow-x-auto pb-1">
        {FILTERS.map((f) => {
          const selected = filter === f.key;
          return (
            <button
              key={f.key}
              type="button"
              onClick={() => setFilter(f.key)}
              aria-pressed={selected}
              className={
                "shrink-0 rounded-full border px-3 py-1.5 text-sm font-medium " +
                (selected
                  ? "border-accent-edge bg-accent-soft text-accent-ink"
                  : "border-border-strong bg-raised text-secondary hover:bg-inset")
              }
            >
              {f.label}
            </button>
          );
        })}
      </div>

        emptyState={
          <EmptyState
            icon={<MapPin />}

                <div className="space-y-2">

                          }

                    })}

                </div>
              }
              trailing={
                <div className="flex flex-col items-end gap-1">
                  <span className="text-sm font-semibold text-primary">{formatMoney(order.total)}</span>
                  {order.status && <Badge tone={tone}>{order.status}</Badge>}

              }
            />
          );
        })}
      </List>
```

## 20. Database schema

products

- name: text
- category: text (Beer or Snacks)

- stock: numeric
- photo: text
- description: text
- created_at: timestamptz, default now()

orders
- id: bigint, primary key, auto-increment
- customer_name: text
- phone: text
- address: text
- delivery_window: text
- notes: text
- total: numeric
- status: text (Pending, Preparing, Out for delivery, Delivered)
- created_at: timestamptz

There are no migration files in the project. To recreate the schema, run CREATE TABLE statements for the two tables above. Then import the CSV files:
- documents/pivas-products-export.csv into products

## 22. Running both apps outside GlideOS

1. Create a Postgres database on your new host. Run the two CREATE TABLE statements from section 20.
2. Import the two CSV files into the tables.
3. Set DATABASE_URL to the new database connection string.
4. Create a Node project. Install the packages from section 19.
5. Mount the server.ts routes in a Hono server, for each app. Serve the built React client from the same server.
6. Replace GlideOS sign-in and access settings with your new host's login. GlideOS sign-in and the profile menu do not carry over.
7. Start each server and open its URL.

Parts that must be rebuilt outside GlideOS: sign-in and the profile menu, the access settings (members only, guests, public), the automatic refresh of data after a change (the platform's query refresh), and the build and preview setup.
